# 可行性：高 ✅
# 工作量：中等（2-3天）

func calculate_memory_metrics(character: String):
    var memories = MemoryManager.get_all_memories(character)
    var total = memories.size()
    var accessed = 0
    var forgotten = 0
    
    for memory in memories:
        if memory["access_count"] > 0:
            accessed += 1
        if memory["decay_factor"] > 0.9:
            forgotten += 1
    
    return {
        "utilization_rate": float(accessed) / total,
        "forgetting_rate": float(forgotten) / total,
        "total_memories": total
    }
```

    var actual_edges = 0

### 挑战3: 个性一致性评估的客观性

**风险等级**: 🔶 中等

# 多维度评估
personality_dimensions = {
    "外向性": ["主动发起对话", "参与群体活动"],
    "开放性": ["接受新想法", "尝试新方法"],
    "责任心": ["按时完成任务", "遵守承诺"],
    "宜人性": ["帮助他人", "避免冲突"],
    "情绪稳定性": ["情绪波动", "压力应对"]
}

# 通过行为模式量化
def quantify_personality(character_behaviors):
    scores = {}
    for dimension, indicators in personality_dimensions.items():
        scores[dimension] = calculate_dimension_score(
            character_behaviors, indicators
        )
    return scores
```

**可行性**: ✅ 有多种评估方法
